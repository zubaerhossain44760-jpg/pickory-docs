<?php
/**
 * includes/services/class-ppfx-product-query-builder.php
 * Orchestrates query compilation and scoped SQL lifecycle for filtered WooCommerce catalog queries.
 */
final class PPFX_Product_Query_Builder {
    private $compiler = null;

    /**
     * Applies compiled index filters to WooCommerce's WP_Query instance.
     * SQL hooks stay attached until this exact query has completed, preventing
     * nested or secondary queries from removing them prematurely.
     */
    public function apply_to_query( \WP_Query $query ): void {
        $compilation = $this->compiler()->compile( $this->get_params() );
        $args        = $compilation['args'];

        // Flag the target query instance to activate hook scope guards
        $query->set( 'ppfx_is_filter_query', true );

        if ( ! empty( $args['tax_query'] ) ) {
            $query->set( 'tax_query', $args['tax_query'] );
        }

        // Attach temporary scoped SQL clauses to the posts_clauses filter
        $scope = $compilation['scope'];
        $scope->attach( $query );
    }
}
/**
 * src/js/utils/url.js & src/js/components/filter-form.js
 * Central transactional write gate enforcing URL as Single Source of Truth (SSOT).
 */
export const commitFilterUpdate = (modifierFn, options = {}) => {
  const { resetPage = true, pushHistory = true, source = "user" } = options;
  const current = PPFX_URL_State.getCanonicalState();
  const next = { ...current };

  // Automatic page reset invariant: facet mutations always revert to page 1
  if (resetPage) {
    delete next.page;
    delete next.paged;
  }

  // Execute caller's transactional mutation on detached state clone
  modifierFn(next);

  if (isAutoFilter || source === "submit" || source === "pagination" || source === "sort" || source === "reset") {
    PPFX_URL_State.setCanonicalState(next, { pushHistory, source });
  }
};

-- Compiled posts_clauses query targeting denormalized flat index tables (ppfxi & ppfxt)
-- Bypasses wp_postmeta joins to achieve sub-50ms query execution across 50,000+ SKUs

SELECT DISTINCT wp_posts.ID
FROM wp_posts
INNER JOIN wp_ppfx_product_index AS ppfxi 
    ON ppfxi.product_id = wp_posts.ID
INNER JOIN wp_ppfx_product_terms AS ppfxt_cat 
    ON ppfxt_cat.product_id = wp_posts.ID 
   AND ppfxt_cat.taxonomy = 'product_cat' 
   AND ppfxt_cat.term_id IN (18, 24)
WHERE wp_posts.post_type = 'product'
  AND wp_posts.post_status = 'publish'
  AND ppfxi.stock_status = 'instock'
  AND ppfxi.min_price >= 25.00
  AND ppfxi.max_price <= 150.00
  AND ppfxi.rating_average >= 4.00
ORDER BY ppfxi.min_price ASC
LIMIT 0, 24;
